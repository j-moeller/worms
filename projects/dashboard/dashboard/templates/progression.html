import json, time, urllib.request

TOKEN = "..."   # a token of your team
BASE = "{{ base }}"
ROUND = "{{ round_id }}"   # a round's id (a uuid): get("/api/v1/rounds/") lists them

def get(path):
    request = urllib.request.Request(BASE + path, headers={"Authorization": f"Bearer {TOKEN}"})
    with urllib.request.urlopen(request, timeout=60) as response:
        return response.read()

def every(path):
    """Every item of a paged list: follow page.next until it is None."""
    items = []
    while path:
        page = json.loads(get(path))
        items += page["items"]
        path = page["page"]["next"]
        time.sleep(0.3)   # stay under the rate limit
    return items

rounds = json.loads(get("/api/v1/rounds/"))["rounds"]
scoring = every(f"/api/v1/rounds/{ROUND}/scoring/?limit=500")
logs = every(f"/api/v1/rounds/{ROUND}/logging/?limit=200")
with open("progression.zip", "wb") as file:
    file.write(get(f"/api/v1/rounds/{ROUND}/progression.zip"))
